FOUNDATIONAL (Challenges 1-3): Below grade level prerequisites
- Tests prerequisite knowledge students SHOULD already know
- Simpler language, concrete examples
- Single-step reasoning
- Example for Grade 5 Fractions: "Which fraction is bigger: 1/2 or 1/4?"

GRADE_LEVEL (Challenges 4-6): Expected mastery for grade {{gradeLevel}}
- Tests the core concepts of {{topic}} at grade level
- Standard curriculum-aligned problems
- Multi-step reasoning may be required
- Example for Grade 5 Fractions: "Add 2/3 + 1/4 and simplify your answer"

ADVANCED (Challenges 7-8+): Above grade level extension
- Tests deeper understanding and application
- Complex problems, multi-step reasoning
- Real-world application or cross-topic synthesis
- Example for Grade 5 Fractions: "A recipe needs 3/4 cup of flour. If you want to make 2.5 batches, how much flour do you need?"

WEIGHTED SCORING:
- Only foundational correct: ~30-40% score
- Foundational + grade-level correct: ~60-75% score
- All tiers correct: ~90-100% score
- This ensures the score maps meaningfully to track placement

<script>
// ========================================
// DIAGNOSTIC STATE (required)
// ========================================
let startTime = Date.now();
let currentChallengeNumber = 1;
let totalChallenges = parseInt(document.querySelector('meta[name="total-challenges"]')?.content || '8');
let challengeStartTime = Date.now();
let challengeResults = []; // Per-challenge performance data

// Tier definitions: maps challenge number to difficulty tier
const challengeTiers = {};
for (let i = 1; i <= totalChallenges; i++) {
    if (i <= 3) challengeTiers[i] = 'foundational';
    else if (i <= 6) challengeTiers[i] = 'grade_level';
    else challengeTiers[i] = 'advanced';
}

// ========================================
// POSTMESSAGE HELPER (required)
// ========================================
function sendToParent(eventData) {
    try {
        if (window.parent && window.parent !== window) {
            window.parent.postMessage({
                type: eventData.type.toLowerCase(),
                data: eventData
            }, '*');
        } else {
            console.log('[DEV] PostMessage:', eventData);
        }
    } catch (error) {
        console.error('PostMessage failed:', error);
        try {
            window.parent.postMessage({
                type: 'quest_error',
                data: {
                    questId: '{{questId}}',
                    errorType: 'postmessage_failed',
                    errorMessage: error.message,
                    recoverable: true,
                    timestamp: Date.now()
                }
            }, '*');
        } catch (e) {
            console.error('Cannot communicate with parent window');
        }
    }
}

// ========================================
// GLOBAL ERROR HANDLER (required)
// ========================================
window.addEventListener('error', (event) => {
    console.error('Diagnostic runtime error:', event.error);
    sendToParent({
        type: 'QUEST_ERROR',
        questId: '{{questId}}',
        errorType: 'runtime_error',
        errorMessage: event.error?.message || 'Unknown error occurred',
        recoverable: false,
        timestamp: Date.now()
    });
});

// ========================================
// DIAGNOSTIC FLOW (required)
// ========================================
function startDiagnostic() {
    document.getElementById('introScreen').classList.remove('active');
    document.getElementById('challengeScreen').classList.add('active');
    startTime = Date.now();
    challengeStartTime = Date.now();
    showChallenge(1);
}

function showChallenge(challengeNum) {
    currentChallengeNumber = challengeNum;
    challengeStartTime = Date.now();

    // Hide all challenge sections, show current one
    document.querySelectorAll('.challenge-section').forEach(el => {
        el.style.display = 'none';
    });
    const currentSection = document.querySelector(`[data-challenge="${challengeNum}"]`);
    if (currentSection) {
        currentSection.style.display = 'block';
    }

    // Update progress bar
    const progressFill = document.querySelector('.progress-fill');
    if (progressFill) {
        progressFill.style.width = `${(challengeNum / totalChallenges) * 100}%`;
    }
    const progressText = document.querySelector('.progress-text');
    if (progressText) {
        progressText.textContent = `Challenge ${challengeNum} of ${totalChallenges}`;
    }

    // Update tier indicator with friendly labels
    const tier = challengeTiers[challengeNum];
    const tierLabel = document.querySelector('.tier-label');
    if (tierLabel) {
        tierLabel.setAttribute('data-tier', tier);
        if (tier === 'foundational') tierLabel.textContent = 'Warm-Up';
        else if (tier === 'grade_level') tierLabel.textContent = 'On Target';
        else tierLabel.textContent = 'Challenge Round';
    }

    // Announce to screen readers
    announceToScreenReader(`Challenge ${challengeNum} of ${totalChallenges}. Difficulty: ${
        tier === 'foundational' ? 'Warm-Up' :
        tier === 'grade_level' ? 'On Target' : 'Challenge Round'
    }`);
}

// Keyboard support: Escape closes modals
document.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') {
        closeModal();
    }
});

function checkAnswer(challengeNum) {
    const isCorrect = validateAnswer(challengeNum); // YOU IMPLEMENT
    const timeOnChallenge = Math.floor((Date.now() - challengeStartTime) / 1000);
    const tier = challengeTiers[challengeNum];

    // Record this challenge's result
    challengeResults.push({
        challengeId: `challenge_${challengeNum}`,
        correct: isCorrect,
        wasSkipped: false,
        attempts: 1,
        timeSpentSeconds: timeOnChallenge,
        difficultyTier: tier
    });

    // REQUIRED: Send CHALLENGE_ATTEMPT event
    sendToParent({
        type: 'CHALLENGE_ATTEMPT',
        questId: '{{questId}}',
        challengeId: `challenge_${challengeNum}`,
        correct: isCorrect,
        attemptNumber: 1,
        selectedAnswer: getSelectedAnswer(challengeNum), // YOU IMPLEMENT
        difficultyTier: tier,
        timestamp: Date.now()
    });

    // Show brief correct/incorrect feedback (NO hints, NO explanation)
    if (isCorrect) {
        showBriefFeedback(true, challengeNum);
    } else {
        showBriefFeedback(false, challengeNum);
    }
}

function skipChallenge(challengeNum) {
    const timeOnChallenge = Math.floor((Date.now() - challengeStartTime) / 1000);
    const tier = challengeTiers[challengeNum];

    // Record skipped challenge
    challengeResults.push({
        challengeId: `challenge_${challengeNum}`,
        correct: false,
        wasSkipped: true,
        attempts: 0,
        timeSpentSeconds: timeOnChallenge,
        difficultyTier: tier
    });

    sendToParent({
        type: 'CHALLENGE_ATTEMPT',
        questId: '{{questId}}',
        challengeId: `challenge_${challengeNum}`,
        correct: false,
        attemptNumber: 0,
        selectedAnswer: null,
        difficultyTier: tier,
        skipped: true,
        timestamp: Date.now()
    });

    announceToScreenReader(`Challenge ${challengeNum} skipped. Moving to next challenge.`);
    advanceToNext(challengeNum);
}

function showBriefFeedback(isCorrect, challengeNum) {
    // Brief feedback only — no teaching, no hints
    const title = isCorrect ? '✓ Correct!' : '✗ Not quite.';
    const content = isCorrect
        ? '<p>Nice work!</p>'
        : '<p>Let\'s keep going.</p>';

    displayModal(title, content, () => advanceToNext(challengeNum));

    announceToScreenReader(isCorrect ? 'Correct!' : 'Incorrect. Moving on.');
}

function advanceToNext(completedChallengeNum) {
    if (completedChallengeNum >= totalChallenges) {
        completeDiagnostic();
    } else {
        showChallenge(completedChallengeNum + 1);
    }
}

    const timeSpent = Math.floor((Date.now() - startTime) / 1000);

// ========================================
// WEIGHTED SCORING (required - CRITICAL)
// ========================================
// Weights: foundational=20%, grade_level=45%, advanced=35%
// This ensures:
//   - Only foundational correct -> ~30-40%
//   - Foundational + grade-level correct -> ~60-75%
//   - All correct -> ~90-100%
function calculateWeightedScore() {
    const tierScores = calculateTierScores();

    const foundationalWeight = 0.20;
    const gradeLevelWeight = 0.45;
    const advancedWeight = 0.35;

    const weightedScore =
        (tierScores.foundational * foundationalWeight) +
        (tierScores.gradeLevel * gradeLevelWeight) +
        (tierScores.advanced * advancedWeight);

    return Math.round(weightedScore);
}

function calculateTierScores() {
    const tiers = { foundational: [], gradeLevel: [], advanced: [] };

    challengeResults.forEach(result => {
        if (result.difficultyTier === 'foundational') {
            tiers.foundational.push(result.correct ? 1 : 0);
        } else if (result.difficultyTier === 'grade_level') {
            tiers.gradeLevel.push(result.correct ? 1 : 0);
        } else if (result.difficultyTier === 'advanced') {
            tiers.advanced.push(result.correct ? 1 : 0);
        }
    });

    const avgScore = (arr) => {
        if (arr.length === 0) return 0;
        return Math.round((arr.reduce((a, b) => a + b, 0) / arr.length) * 100);
    };

    return {
        foundational: avgScore(tiers.foundational),
        gradeLevel: avgScore(tiers.gradeLevel),
        advanced: avgScore(tiers.advanced)
    };
}

function displayModal(title, content, onClose) {
    const existingModal = document.getElementById('feedbackModal');

    const modal = document.createElement('div');
    modal.id = 'feedbackModal';
    modal.setAttribute('role', 'dialog');

    modal.setAttribute('aria-labelledby', 'modalTitle');
    modal.style.cssText = `
        position: fixed; top: 0; left: 0; width: 100%; height: 100%;
        background: rgba(0,0,0,0.8); display: flex;
        align-items: center; justify-content: center;
        z-index: 10000; animation: fadeIn 0.3s ease;
    `;

    modal.innerHTML = `
        <div class="modal-content" style="
            background: white; border-radius: 20px; padding: 40px;
            max-width: 400px; text-align: center; animation: slideIn 0.3s ease;">
            <h2 id="modalTitle" style="font-family: var(--font-primary); margin-bottom: 15px;">${title}</h2>
            <div id="modalContent" style="margin: 15px 0;">${content}</div>
            <button onclick="closeModal()"
                    aria-label="Continue to next challenge"
                    style="margin-top: 15px; padding: 12px 24px;
                        background: var(--color-primary); color: white;
                        border: none; border-radius: 12px;
                        font-family: var(--font-primary); font-size: 16px;
                        cursor: pointer;">
                Continue →
            </button>
        </div>
    `;

    // Store onClose callback
    modal._onClose = onClose;
    document.body.appendChild(modal);
    modal.querySelector('button').focus();

    // Add CSS animations if not already present
    if (!document.getElementById('modalAnimations')) {
        const style = document.createElement('style');
        style.id = 'modalAnimations';
        style.textContent = `
            @keyframes fadeIn {
                from { opacity: 0; }
                to { opacity: 1; }
            }
            @keyframes slideIn {
                from { transform: translateY(-50px); opacity: 0; }
                to { transform: translateY(0); opacity: 1; }
            }
        `;
        document.head.appendChild(style);
    }
}

function closeModal() {
    const modal = document.getElementById('feedbackModal');
    if (modal) {
        const onClose = modal._onClose;
        modal.remove();
        if (typeof onClose === 'function') onClose();
    }
}
