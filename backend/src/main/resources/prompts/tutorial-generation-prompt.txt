<script>
// ========================================
// TUTORIAL STATE (required)
// ========================================
let startTime = Date.now();
let currentConcept = 1;
let totalConcepts = 0; // Set from wrongChallenges length or 3 if empty
let tutorialAttempt = parseInt(document.querySelector('meta[name="tutorial-attempt"]')?.content || '1');

// Scoring state — tracks practice problem results
let totalPracticeProblems = 0;   // Total practice problems across all concepts
let correctPracticeProblems = 0; // Number answered correctly (best attempt per problem)
let practiceResults = [];         // Array of {conceptId, problemNumber, correct, attempts, studentAnswer, correctAnswer}

// Per-concept tracking — how many problems attempted in current concept
let practiceAttemptsInConcept = {};  // { conceptNumber: Set of problemNumbers attempted }
let practiceRequiredInConcept = {};  // { conceptNumber: total problems in this concept }

// ========================================
// POSTMESSAGE HELPER (required)
// ========================================
function sendToParent(eventData) {
    try {
        if (window.parent && window.parent !== window) {
            window.parent.postMessage({
                type: eventData.type.toLowerCase(),
                data: eventData
            }, '*');
        } else {
            console.log('[DEV] PostMessage:', eventData);
        }
    } catch (error) {
        console.error('PostMessage failed:', error);
        try {
            window.parent.postMessage({
                type: 'tutorial_error',
                data: {
                    questId: '{{questId}}',
                    errorType: 'postmessage_failed',
                    errorMessage: error.message,
                    recoverable: true,
                    timestamp: Date.now()
                }
            }, '*');
        } catch (e) {
            console.error('Cannot communicate with parent window');
        }
    }
}

// ========================================
// GLOBAL ERROR HANDLER (required)
// ========================================
window.addEventListener('error', (event) => {
    console.error('Tutorial runtime error:', event.error);
    sendToParent({
        type: 'TUTORIAL_ERROR',
        questId: '{{questId}}',
        errorType: 'runtime_error',
        errorMessage: event.error?.message || 'Unknown error occurred',
        recoverable: false,
        timestamp: Date.now()
    });
});

// ========================================
// TUTORIAL FLOW (required)
// ========================================
function startTutorial() {
    document.getElementById('introScreen').classList.remove('active');
    document.getElementById('tutorialScreen').classList.add('active');
    startTime = Date.now();
    showConcept(1);
}

function showConcept(conceptNumber) {
    currentConcept = conceptNumber;

    // Hide all concept sections, show the current one
    document.querySelectorAll('.concept-section').forEach(s => s.style.display = 'none');
    const section = document.getElementById('concept-' + conceptNumber);
    if (section) section.style.display = 'block';

    // Update progress indicator
    const progress = document.getElementById('conceptProgress');
    if (progress) {
        progress.textContent = `Concept ${conceptNumber} of ${totalConcepts}`;
    }

    // Disable "Next Concept" until all practice problems are attempted
    updateNextButton();

    // Announce to screen readers
    announceToScreenReader(`Now reviewing concept ${conceptNumber} of ${totalConcepts}`);
}

function updateNextButton() {
    const btn = document.getElementById('nextConceptBtn');
    if (!btn) return;

    const required = practiceRequiredInConcept[currentConcept] || 0;
    const attempted = practiceAttemptsInConcept[currentConcept]
        ? practiceAttemptsInConcept[currentConcept].size : 0;

    if (attempted >= required) {
        btn.disabled = false;
        btn.textContent = currentConcept < totalConcepts ? 'Next Concept' : 'See Results';
        btn.setAttribute('aria-label',
            currentConcept < totalConcepts ? 'Continue to next concept' : 'See your results');
    } else {
        btn.disabled = true;
        btn.textContent = `Answer ${required - attempted} more to continue`;
        btn.setAttribute('aria-label',
            `Answer ${required - attempted} more practice problems to continue`);
    }
}

function nextConcept() {
    if (currentConcept < totalConcepts) {
        showConcept(currentConcept + 1);
    } else {
        completeTutorial();
    }
}

// ========================================
// PRACTICE PROBLEM CHECKING (required)
// ========================================
// Practice problems ARE scored. Feedback is immediate and explanatory.
function checkPractice(conceptNumber, problemNumber) {
    const result = validatePractice(conceptNumber, problemNumber); // YOU IMPLEMENT
    // result must be: { correct: boolean, studentAnswer: string, correctAnswer: string }

    // Track attempt
    if (!practiceAttemptsInConcept[conceptNumber]) {
        practiceAttemptsInConcept[conceptNumber] = new Set();
    }
    const isFirstAttempt = !practiceAttemptsInConcept[conceptNumber].has(problemNumber);

    practiceAttemptsInConcept[conceptNumber].add(problemNumber);

    // Record result (only first attempt counts for score)
    if (isFirstAttempt) {
        if (result.correct) correctPracticeProblems++;
        practiceResults.push({
            conceptId: conceptNumber,
            problemNumber: problemNumber,
            correct: result.correct,
            studentAnswer: result.studentAnswer,
            correctAnswer: result.correctAnswer
        });
    }

    if (result.correct) {
        showPracticeFeedback(conceptNumber, problemNumber, true,
            "Correct! " + (result.explanation || ""));
    } else {
        showPracticeFeedback(conceptNumber, problemNumber, false,
            "Not right. " + (result.explanation || "Here's why..."));
        // Show detailed re-explanation for wrong answers
    }

    // Update next button state
    updateNextButton();

    announceToScreenReader(result.correct ?
        "Correct! Well done." :
        "Not right. Read the feedback for an explanation.");
}

function showPracticeFeedback(conceptNumber, problemNumber, isCorrect, message) {
    const feedbackEl = document.getElementById(
        `feedback-${conceptNumber}-${problemNumber}`
    );
    if (feedbackEl) {
        feedbackEl.style.display = 'block';
        feedbackEl.innerHTML = isCorrect
            ? `<div style="color: var(--color-correct); padding: 12px; border-radius: 8px; background: #d4edda;">
                 &#10003; ${message}
               </div>`
            : `<div style="color: var(--color-incorrect); padding: 12px; border-radius: 8px; background: #ffeaa7;">
                 &#10148; ${message}
               </div>`;
    }
}

// TUTORIAL COMPLETION (required)
// ========================================
function completeTutorial() {
    const timeSpent = Math.floor((Date.now() - startTime) / 1000);
    const score = totalPracticeProblems > 0
        ? Math.round((correctPracticeProblems / totalPracticeProblems) * 100)
        : 0;

    // REQUIRED: Send TUTORIAL_COMPLETE event WITH score, challenge results, AND attempt number
    sendToParent({
        type: 'TUTORIAL_COMPLETE',
        questId: '{{questId}}',
        score: score,
        tutorialAttempt: tutorialAttempt,
        timeSpent: timeSpent,
        conceptsReviewed: totalConcepts,
        totalChallenges: totalPracticeProblems,
        completedChallenges: correctPracticeProblems,
        skippedChallenges: 0,
        challengeResults: practiceResults,
        timestamp: Date.now()
    });

    // Show results screen
    document.getElementById('tutorialScreen').classList.remove('active');
    document.getElementById('celebrationScreen').classList.add('active');

    // Populate results
    const scoreDisplay = document.getElementById('scoreDisplay');
    if (scoreDisplay) {
        scoreDisplay.innerHTML = `<p style="font-size: 1.5em; font-family: var(--font-primary);">
            You got <strong>${correctPracticeProblems}</strong> out of
            <strong>${totalPracticeProblems}</strong> practice problems right.
        </p>`;
    }

    const resultTitle = document.getElementById('resultTitle');
    const resultMessage = document.getElementById('resultMessage');

    if (score >= 80) {
        if (resultTitle) resultTitle.textContent = 'You\'ve Got This!';
        if (resultMessage) resultMessage.innerHTML = `<p>You clearly understand these concepts now.
            That's real progress — nice work putting in the effort.</p>`;
    } else if (score >= 50) {
        if (resultTitle) resultTitle.textContent = 'Getting Stronger';
        if (resultMessage) resultMessage.innerHTML = `<p>You're building understanding.
            Keep these concepts in mind as you move forward.</p>`;
    } else if (tutorialAttempt === 1) {
        // Low score on first attempt — a retry tutorial will be generated
        if (resultTitle) resultTitle.textContent = 'Let\'s Try a Different Approach';
        if (resultMessage) resultMessage.innerHTML = `<p>These concepts are tricky.
            A new review is being prepared that breaks things down further.
            Different explanations work for different people — let's find what clicks for you.</p>`;
    } else {
        // Low score on second attempt — direct to teacher
        if (resultTitle) resultTitle.textContent = 'Keep Working At It';
        if (resultMessage) resultMessage.innerHTML = `<p>These concepts take time.
            Talk to your teacher if you'd like to go over them together — that's what they're there for.</p>`;
    }

    playCelebration(); // YOU IMPLEMENT: celebration animation proportional to score
}

            <div id="modalContent" style="margin: 20px 0;">${content}</div>
            <button onclick="closeModal()" aria-label="Close" style="
                margin-top: 20px; padding: 12px 24px;
                background: var(--color-primary); color: white;
                border: none; border-radius: 12px;
                font-family: var(--font-primary); font-size: 16px;
                cursor: pointer;">
                Got It!
            </button>
        </div>
    `;

    document.body.appendChild(modal);
    modal.querySelector('button').focus();

    if (!document.getElementById('modalAnimations')) {
        const style = document.createElement('style');
        style.id = 'modalAnimations';
        style.textContent = `
            @keyframes fadeIn { from { opacity: 0; } to { opacity: 1; } }
            @keyframes slideIn { from { transform: translateY(-50px); opacity: 0; } to { transform: translateY(0); opacity: 1; } }
        `;
        document.head.appendChild(style);
    }
}
